Window Function — Sales Ranking by Region
-- Rank products by sales within each region
SELECT
  region,
  product_name,
  sales_amount,
  RANK() OVER (
    PARTITION BY region
    ORDER BY sales_amount DESC
  ) AS sales_rank,
  SUM(sales_amount) OVER (
    PARTITION BY region
  ) AS region_total
FROM sales_data
WHERE fiscal_year = 2025
ORDER BY region, sales_rank;
PL/SQL Procedure — Data Validation & Error Handling
CREATE OR REPLACE PROCEDURE
  validate_sales_data(
    p_batch_id IN NUMBER
  ) IS
  v_count NUMBER;
BEGIN
  SELECT COUNT(*) INTO v_count
  FROM staging_sales
  WHERE batch_id = p_batch_id
  AND sales_amount IS NULL;
  IF v_count > 0 THEN
    RAISE_APPLICATION_ERROR(
      -20001, 'Null values in batch');
  END IF;
EXCEPTION WHEN OTHERS THEN
  DBMS_OUTPUT.PUT_LINE(SQLERRM);
END;
Subquery — Top Performers per Department
-- Find employees above dept average salary
SELECT e.emp_name, e.department, e.salary,
  dept_avg.avg_sal
FROM employees e
JOIN (
  SELECT department,
    AVG(salary) AS avg_sal
  FROM employees
  GROUP BY department
) dept_avg
  ON e.department = dept_avg.department
WHERE e.salary > dept_avg.avg_sal
ORDER BY e.department, e.salary DESC;
ETL — Data Load with Merge Statement
-- Upsert: insert new, update existing
MERGE INTO target_sales t
USING staging_sales s
  ON (t.sale_id = s.sale_id)
WHEN MATCHED THEN
  UPDATE SET
    t.amount = s.amount,
    t.updated_at = SYSDATE
WHEN NOT MATCHED THEN
  INSERT (sale_id, amount, region,
    created_at)
  VALUES (s.sale_id, s.amount,
    s.region, SYSDATE);
